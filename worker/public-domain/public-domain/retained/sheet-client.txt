  if(historyBody.childElementCount)meta?.append(history);
  const save=editor.querySelector('.cvt-savebar');if(save)meta?.append(save);
  const options=actions?.querySelector('.cvt-document-options>summary');if(options){options.textContent='';options.append(svg('more'));options.setAttribute('aria-label','Document options');options.title='Document options';options.classList.add('cvs-icon-control')}

   else body.append(make('p','','Recorded history is unavailable.'));
  }
  function selectCell(cell){if(!cell)return;selectedKey=cell.dataset.cell;for(const c of sheet.querySelectorAll('.cvs-selected-cell'))if(c!==cell)c.classList.remove('cvs-selected-cell');cell.classList.add('cvs-selected-cell');signature='';refreshInspector()}

 }
