(()=>{

  const form=document.querySelector('#cv-ws-request-form');
  if(!form)return;
  const select=form.querySelector('select[name="kind"]'),tiles=form.querySelector('.cv3-kind-selector');
  if(!select||!tiles||select.dataset.cv9Type)return;

 }
