(()=>{'use strict';const seed=document.getElementById('cv-discovery-config');if(!seed)return;const config=JSON.parse(seed.textContent),state={events:null,clubs:null,feed:null};let frame=0;
const make=(tag,cls,text)=>{const e=document.createElement(tag);if(cls)e.className=cls;if(text)e.textContent=text;return e;};
function signIn(parent){const a=make('a','cv-glass-link','Sign in');a.href='/sign-in/';parent.append(a);}
function notice(id,grid,message,kind){let e=document.getElementById(id);if(!e){e=make('section','cv-empty-state');e.id=id;e.setAttribute('role','status');grid.after(e);}if(e.dataset.message!==message){e.replaceChildren(make('p','',message));if(kind==='signin')signIn(e);else if(kind==='clubs'){const b=make('button','cv-glass-link','Create a club');b.type='button';b.addEventListener('click',()=>{const original=[...document.querySelectorAll('button')].find(v=>v!==b&&(/Create a club/i.test(v.textContent)||v.hasAttribute('data-circle-create')));if(original)original.click();else location.assign('/clubs/?create=1');});e.append(b);}else if(kind==='events'){const reset=make('a','cv-glass-link','Reset filters');reset.href='/events/';const workshop=make('a','cv-glass-link','Office workshops');workshop.href='/workshops/';const mail=make('a','cv-glass-link','hello@communiverseclubs.com');mail.href='mailto:hello@communiverseclubs.com';e.append(reset,workshop,mail);}e.dataset.message=message;}e.hidden=false;return e;}
function hide(id){const e=document.getElementById(id);if(e)e.hidden=true;}
function update(){frame=0;for(const select of document.querySelectorAll('#cv-feed-region,#cv-circle-filters select[name=region],#cv-circle-form select[name=region]')){const allowed=select.id==='cv-feed-region'?config.feedRegions:config.clubRegions;for(const o of [...select.options])if(o.value&&!allowed.includes(o.value))o.remove();}
 for(const e of document.querySelectorAll('[data-heart-count]')){const value=Number(e.textContent.trim());e.hidden=value<config.heartThreshold;if(!e.hidden)e.setAttribute('aria-label',value+' hearts');}
 for(const e of document.querySelectorAll('span,p')){const m=e.textContent.trim().match(/^(\d+) hearts(?: on their work)?$/i);if(m){e.classList.add('cv-low-count');e.hidden=Number(m[1])<config.heartThreshold;if(!e.hidden)e.setAttribute('aria-label',m[1]+' hearts');}}
 const mode=document.querySelector('[data-feed-mode][aria-pressed=true]')?.dataset.feedMode,feed=document.getElementById('cv-feed');if(feed&&['saved','following'].includes(mode)&&!state.signedIn){notice('cv-feed-empty',feed,'Sign in to see artists you follow and works you save.','signin');const status=document.getElementById('cv-feed-status');if(status&&status.textContent)status.textContent='';}else hide('cv-feed-empty');
 const events=document.getElementById('cv-events');if(events&&state.events){const d=state.events,overall=d.overallTotal??config.programmeCount,empty=Number(d.total)===0;if(empty){notice('cv-events-empty',events,overall?'No events match these filters. Try another date or reset your filters.':'No events listed yet. Our first Communiverse events in the UAE are coming soon.','events');if(events.childNodes.length)events.replaceChildren();}else hide('cv-events-empty');const controls=document.getElementById('cv-events-filters');if(controls)controls.hidden=overall===0;for(const a of document.querySelectorAll('#cv-social-header nav a'))if(new URL(a.href).pathname==='/events/')a.hidden=overall===0;}
 const clubs=document.getElementById('cv-circles');if(clubs&&state.clubs){const d=state.clubs,following=document.querySelector('[data-circle-view=following]')?.getAttribute('aria-pressed')==='true';if(following&&!state.signedIn)notice('cv-clubs-empty',clubs,'Sign in to see the clubs you follow.','signin');else if(Number(d.total)===0)notice('cv-clubs-empty',clubs,(d.overallTotal??config.clubCount)?'No clubs match these filters. Try another interest or choose Everywhere.':'No clubs yet. Start the first one.','clubs');else hide('cv-clubs-empty');}
}
function schedule(){if(!frame)frame=requestAnimationFrame(update);}
window.CVDiscovery={response(path,data){const type=path.split('?')[0];if(type==='me')state.signedIn=!!data.member;if(type==='events'&&Array.isArray(data.events)){const active=document.querySelector('[data-event-state][aria-pressed=true]')?.dataset.eventState;if(active&&data.state&&active!==data.state&&!(active==='now'&&data.state==='today'))return;const q=new URLSearchParams(path.split('?')[1]||''),form=document.getElementById('cv-events-filters');for(const name of ['q','region','type','gallery'])if(form?.elements[name]&&String(form.elements[name].value)!==String(q.get(name)||''))return;state.events=data;}if(type==='circles'&&Array.isArray(data.circles))state.clubs=data;if(type==='feed'&&Array.isArray(data.items))state.feed=data;schedule();}};
const observer=new MutationObserver(schedule);observer.observe(document.body,{childList:true,subtree:true,characterData:true});update();addEventListener('pagehide',()=>{observer.disconnect();if(frame)cancelAnimationFrame(frame);});})();
