  function starPositions(
    items,
    w,
    initialH,
    d,
    gap,
    zones = [],
    ellipse = true,
  ) {
    const diameters = items.map((n) => d ?? n.restD);
    const key = JSON.stringify([
      items.map((n) => n.p.slug),
      Math.round(w),
      initialH,
      diameters,
      gap,
      zones,
      ellipse,
    ]);
    if (galaxyCache.has(key)) return galaxyCache.get(key);
    let h = initialH;
    for (let round = 0; round < 24; round++, h = Math.ceil(h * 1.08)) {
      const points = [];
      let failed = false;
      for (const n of items) {
        const diameter = d ?? n.restD,
          r = diameter / 2,
          rx = w / 2 - r - 12,
          ry = h / 2 - r - 12;
        let seed = (n.seed ^ constellationSeed ^ 0x9e3779b9) || 0x6d2b79f5,
          point = null;
        const rand = () => {
          seed ^= seed << 13;
          seed ^= seed >>> 17;
          seed ^= seed << 5;
          return (seed >>> 0) / 4294967296;
        };
        for (let k = 0; k < 1800; k++) {
          const angle = rand() * Math.PI * 2,
            anchor = points.length
              ? points[Math.floor(rand() * points.length)]
              : null,
            distance = anchor
              ? (anchor.d + diameter) / 2 + gap + rand() * 10
              : 0,
            radial = Math.sqrt(rand()) * (k < 900 ? 0.45 : 1);
          const x = anchor
            ? anchor.x + Math.cos(angle) * distance
            : w / 2 + Math.cos(angle) * radial * rx;
          const y = anchor
            ? anchor.y + Math.sin(angle) * distance
            : h / 2 + Math.sin(angle) * radial * ry;
          if (
            x < r + 12 ||
            x > w - r - 12 ||
            y < r + 12 ||
            y > h - r - 12 ||
            (ellipse &&
              ((x - w / 2) / rx) ** 2 + ((y - h / 2) / ry) ** 2 > 1)
          )
            continue;
          if (
            zones.some(
              (z) =>
                x + r + gap > z.x &&
                x - r - gap < z.x + z.w &&
                y + r + gap > z.y &&
                y - r - gap < z.y + z.h,
            )
          )
            continue;
          if (
            points.some(
              (p) => Math.hypot(p.x - x, p.y - y) < (p.d + diameter) / 2 + gap,
            )
          )
            continue;
          point = { x, y, d: diameter };
          break;
        }
        if (!point) {
          failed = true;
          break;
        }
        points.push(point);
      }
      if (!failed) {
        const minX=Math.min(...points.map(p=>p.x-p.d/2)),maxX=Math.max(...points.map(p=>p.x+p.d/2));
        const minY=Math.min(...points.map(p=>p.y-p.d/2)),maxY=Math.max(...points.map(p=>p.y+p.d/2));
        const dx=w/2-(minX+maxX)/2,dy=h/2-(minY+maxY)/2;
        points.forEach(p=>{p.x+=dx;p.y+=dy;});
        const result = { h, points };
        galaxyCache.set(key, result);
        return result;
      }
    }
    throw Error("Cannot fit the public portraits in the constellation");
  }
  function nearbyPositions(items, w, zones) {
    const placed = new Map(), positions = [], r = 22, gap = 10;
    const inZone = (x, y) => zones.some((z) =>
      x + r + gap > z.x && x - r - gap < z.x + z.w &&
      y + r + gap > z.y && y - r - gap < z.y + z.h,
    );
    const free = (x, y) => x >= r + 8 && x <= w - r - 8 && y >= r + 8 &&
      !inZone(x, y) && !positions.some((p) => Math.hypot(p.x - x, p.y - y) < 2 * r + gap);
    const add = (n, p) => { placed.set(n, p); positions.push(p); };
    const displaced = [];
    for (const n of items) {
      if (free(n.rest.x, n.rest.y)) add(n, { ...n.rest, d: 44 });
      else displaced.push(n);
    }
    for (const n of displaced) {
      let destination = null;
      const phase = ((n.seed ^ constellationSeed) >>> 0) / 4294967296 * 2 * Math.PI;
      for (let distance = 12; !destination; distance += 12) {
        for (let k = 0; k < 64; k++) {
          const angle = phase + k * Math.PI / 32;
          const x = n.rest.x + Math.cos(angle) * distance;
          const y = n.rest.y + Math.sin(angle) * distance;
          if (free(x, y)) { destination = { x, y, d: 44 }; break; }
        }
      }
      add(n, destination);
    }
    // Shorten affected journeys by exchanging only their free destinations.
    // Intersecting straight paths cannot be a minimum-distance pairing.
    const cost = (n, p) => {
      const from = focusOrigins?.get(n) || n.rest;
      return Math.hypot(from.x - p.x, from.y - p.y);
    };
    for (let pass = 0; pass < displaced.length; pass++) {
      let changed = false;
      for (let i = 0; i < displaced.length; i++)
        for (let j = i + 1; j < displaced.length; j++) {
          const a = displaced[i], b = displaced[j];
          const pa = placed.get(a), pb = placed.get(b);
          if (cost(a, pb) + cost(b, pa) < cost(a, pa) + cost(b, pb) - 0.1) {
            placed.set(a, pb); placed.set(b, pa); changed = true;
          }
        }
      if (!changed) break;
    }
    return placed;
  }

  function tick(t) {
    frame = 0;
    if (document.hidden) return;
    if (motionDirty || !motionPlan) planMotion();
    if (!motionStart) motionStart = t;
    const elapsed = t - motionStart,
      u = Math.min(1, elapsed / MOTION.duration);
    // Smooth start and finish. Peers make room early; the focus grows continuously.
    const p = u * u * u * (u * (u * 6 - 15) + 10),
      fade = Math.min(1, elapsed / MOTION.fadeDuration);
    orbitProgress=p;
    for (const a of motionPlan) {
      a.n.x = a.sx + (a.tx - a.sx) * p;
      a.n.y = a.sy + (a.ty - a.sy) * p;
      const sizeU = a.focus ? u : Math.min(1, u / (a.returning ? 0.25 : 0.4));
      const sizeP = sizeU * sizeU * sizeU * (sizeU * (sizeU * 6 - 15) + 10);
      a.n.d = a.sd + (a.td - a.sd) * sizeP;
      a.n.o = a.so + (a.to - a.so) * fade;
    }
    if (selected && u >= 0.6) root.classList.add("is-details-ready");
    resolveFrame();
    root.dataset.animating = String(u < 1);
    root.classList.toggle("is-animating", u < 1);
    if (u < 1) frame = requestAnimationFrame(tick);
    else snap();
  }
  function wake() {
    if (document.hidden) return;
    if (reduced.matches) {
      cancelAnimationFrame(frame);
      frame = 0;
      snap();
      return;
    }
    if (motionDirty) planMotion();
    if (!frame && motionPlan) frame = requestAnimationFrame(tick);
  }
  for (const b of buttons)
    b.addEventListener(
      "click",
      () => {
        if (selected) close(false);
        filter = b.dataset.filter;
        for (const x of buttons)
          x.setAttribute("aria-pressed", String(x === b));
        const u = new URL(location.href);
        if (filter === "communiverse") u.searchParams.delete("filter");
        else u.searchParams.set("filter", filter);
        u.searchParams.delete("person");
        writePage(u);
        history.replaceState(null, "", u);
        measure();
        targets();
        wake();
      },
      { signal },
    );
  function writePage(u) {
    u.searchParams.delete("page");
  }
  function restoreLocation() {
    const u = new URL(location.href),
      f = normalizeFilter(u.searchParams.get("filter"));
    panel.querySelectorAll("video").forEach(v=>v.pause());
    rebase();
    selected = null;
    focusAnchor = focusOrigins = null;
    root.classList.remove("is-focused", "is-details-ready");
    delete root.dataset.focusedPerson;
    panel.inert = true;
    panel.setAttribute("aria-hidden", "true");
    identity.setAttribute("aria-hidden", "true");
    panelAnimation?.cancel();
    filter = buttons.some((b) => b.dataset.filter === f) ? f : "communiverse";
    buttons.forEach((b) =>
      b.setAttribute("aria-pressed", String(b.dataset.filter === filter)),
    );
    measure();
    const n = nodes.find((n) => n.p.slug === u.searchParams.get("person")),
      i = allMatching.indexOf(n);
    if (n && i >= 0) {
      measure();
      select(n, false);
    } else {
      targets();
      wake();
    }
    if (document.activeElement?.matches('.node[aria-hidden="true"]'))
      (visible[0]?.el || buttons[0]).focus({ preventScroll: true });
  }
  panel.addEventListener(
    "focusin",
    () => {
      if (selected && !root.classList.contains("is-details-ready")) {
        cancelAnimationFrame(frame);
        frame = 0;
        snap();
      }
    },
    { signal },
  );
  window.addEventListener("popstate", restoreLocation, { signal });
  document.addEventListener(
    "keydown",
    (e) => {
      if (e.key === "Escape" && selected) {
        e.preventDefault();
        close(true);
      }
    },
    { signal },
  );
  stage.addEventListener(
    "click",
    (e) => {
      if (selected && e.target === stage) {
        const expanded=panel.querySelector('.is-expanded');
        if(expanded){expanded.classList.remove('is-expanded');const extra=expanded.querySelector('details');if(extra)extra.open=false;}
        else close(false);
      }
    },
    { signal },
  );
  document.addEventListener('pointerdown',e=>{
    const expanded=panel.querySelector('.is-expanded');
    if(expanded&&!expanded.contains(e.target)){expanded.classList.remove('is-expanded');const extra=expanded.querySelector('details');if(extra)extra.open=false;}
  },{signal});
  // One lifecycle owns all frame work. BFCache retains listeners; real teardown releases them.
  const ro = new ResizeObserver(() => {
    cancelAnimationFrame(resizeFrame);
    resizeFrame = requestAnimationFrame(() => {
      measure();
      targets();
      wake();
    });
  });
  ro.observe(stage);
  ro.observe(panel);
  function stopFrames() {
    cancelAnimationFrame(frame);
    cancelAnimationFrame(resizeFrame);
    frame = resizeFrame = 0;
    panelAnimation?.cancel();
  }
  reduced.addEventListener(
    "change",
    () => {
      panelAnimation?.cancel();
      if(reduced.matches)panel.querySelectorAll('video').forEach(v=>v.pause());
      wake();
    },
    { signal },
  );
  document.addEventListener(
    "visibilitychange",
    () => {
      if (document.hidden) {stopFrames();panel.querySelectorAll('video').forEach(v=>v.pause());}
      else {
        measure();
        targets();
        wake();
      }
    },
    { signal },
  );
  window.addEventListener(
    "resize",
    () => {
      cancelAnimationFrame(resizeFrame);
      resizeFrame = requestAnimationFrame(() => {
        measure();
        targets();
        wake();
      });
    },
    { signal },
  );
  window.addEventListener(
    "pagehide",
    (e) => {
      stopFrames();
      if (!e.persisted) {
        ro.disconnect();
        lifetime.abort();
      }
    },
    { signal },
  );
  window.addEventListener(
    "pageshow",
    () => {
      measure();
      targets();
      wake();
    },
    { signal },
  );
  for (const b of buttons)
    b.setAttribute("aria-pressed", String(b.dataset.filter === filter));
  const currentUrl = new URL(location.href);
  if (currentUrl.searchParams.has("page")) {
    currentUrl.searchParams.delete("page");
    history.replaceState(null, "", currentUrl);
  }
  measure();
  targets();
  snap();
  root.dataset.glassReady = "true";
  const initialPerson = new URL(location.href).searchParams.get("person");
  if (initialPerson) {
    const n = nodes.find((n) => n.p.slug === initialPerson);
    if (n) {
      const i = allMatching.indexOf(n);
      if (i >= 0) {
        measure();
        select(n, false);
        if (reduced.matches) snap();
      }
    }
  }

  window.__cvPlugFocusState = () => ({
    release: RELEASE,
    selected: selected?.p.slug || null,
    filter,
    pageIndex: 0,
    totalPages: 1,
    totalMatching: allMatching.length,
    visible: visible.map((n) => n.p.slug),
    animating: !!frame,
    residual: Math.max(
      0,
      ...visible.map((n) => Math.hypot(n.x - n.tx, n.y - n.ty)),
    ),
    reducedMotion: reduced.matches,
  });
})();
